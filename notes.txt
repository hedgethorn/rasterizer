Scalar renderer for reference

void render_scalar(
	struct Transform transform,
	u32 *faces,
	u32 face_count,
	f32 *untransformed_vertexes,
	u32 vertex_count,
	u32 *frame,
	f32 *z_buffer,
	u32 width,
	u32 height
) {
	struct ThreadContext *lx = thread_context();

	f32 *vertexes = arena_alloc(&lx->scratch, vertex_count * 12);

	for (u32 i = 0; i < vertex_count; i++) {
		f32 x = untransformed_vertexes[i + vertex_count * 0];
		f32 y = untransformed_vertexes[i + vertex_count * 1];
		f32 z = untransformed_vertexes[i + vertex_count * 2];

		v3 t = matmul(transform.matrix, (v3){x, y, z});
		t.x += transform.translation[0];
		t.y += transform.translation[1];
		t.z += transform.translation[2];

		t.x = t.x / t.z;
		t.y = t.y / t.z;
		t.z = 1.0 / t.z;

		vertexes[i + vertex_count * 0] = t.x;
		vertexes[i + vertex_count * 1] = t.y;
		vertexes[i + vertex_count * 2] = t.z;
	}

	for (u32 y = 0; y < height; y++) {
		for (u32 x = 0; x < width; x++) {
			f32 px = ((f32)x / (f32)width) * 2.0f - 1.0f;
			f32 py = ((f32)y / (f32)height) * 2.0f - 1.0f;

			for (u32 i = 0; i < face_count; i++) {
				u32 ai = faces[i + face_count * 0];
				u32 bi = faces[i + face_count * 1];
				u32 ci = faces[i + face_count * 2];

				f32 ax = vertexes[ai + vertex_count * 0];
				f32 ay = vertexes[ai + vertex_count * 1];
				f32 az = vertexes[ai + vertex_count * 2];

				f32 bx = vertexes[bi + vertex_count * 0];
				f32 by = vertexes[bi + vertex_count * 1];
				f32 bz = vertexes[bi + vertex_count * 2];

				f32 cx = vertexes[ci + vertex_count * 0];
				f32 cy = vertexes[ci + vertex_count * 1];
				f32 cz = vertexes[ci + vertex_count * 2];

				f32 abc = edge(v2f(ax, ay), v2f(bx, by), v2f(cx, cy));
				f32 abp = edge(v2f(ax, ay), v2f(bx, by), v2f(px, py));
				f32 bcp = edge(v2f(bx, by), v2f(cx, cy), v2f(px, py));
				f32 cap = edge(v2f(cx, cy), v2f(ax, ay), v2f(px, py));

				f32 c = abp / abc;
				f32 a = bcp / abc;
				f32 b = cap / abc;

				f32 z = 1.0f / (a * az + b * bz + c * cz);

				if (z < z_buffer[y * width + x] && 0 < abp && 0 < bcp && 0 < cap) {
					u32 color = (u32)(((f32)i / (f32)face_count) * 200.0) + 50;
					frame[y * width + x] = color << 16 | color << 8 | color << 0;
					z_buffer[y * width + x] = z;
				}
			}
		}
	}
}
